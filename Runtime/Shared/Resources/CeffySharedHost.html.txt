<head>
  <meta charset="utf-8" />
  <title>Ceffy Shared Host</title>
  <style>
    html, body {
      margin: 0;
      padding: 0;
      width: 100%;
      height: 100%;
      overflow: hidden;
      background: transparent;
    }
    #root {
      position: relative;
      width: 100%;
      height: 100%;
      background: transparent;
    }
    iframe.ceffy-slot {
      position: absolute;
      border: 0;
      margin: 0;
      padding: 0;
      background: transparent;
      overflow: hidden;
      display: block;
    }
  </style>
</head>
<body>
  <div id="root"></div>
  <script>
(function () {
  const root = document.getElementById("root");
  /** @type {Map<string, {iframe: HTMLIFrameElement, pending: string[]}>} */
  const slots = new Map();

  const consoleLevels = { debug: 1, log: 2, info: 2, warn: 3, error: 4 };
  let pendingTimer = 0;

  function sendToUnity(payload) {
    window.ceffy.SendToUnity(JSON.stringify(payload));
  }

  function stringify(value) {
    if (typeof value === "string") return value;
    try { return JSON.stringify(value); } catch { return String(value); }
  }

  function forwardConsole(id, iframe) {
    const win = iframe.contentWindow;
    Object.keys(consoleLevels).forEach(function (method) {
      const original = win.console[method];
      win.console[method] = function () {
        const text = Array.prototype.map.call(arguments, stringify).join(" ");
        sendToUnity({ type: "console", id: id, level: consoleLevels[method], data: text, src: win.location.href });
        original.apply(win.console, arguments);
      };
    });
  }

  function flushPending() {
    let waiting = false;
    slots.forEach(function (slot) {
      if (!slot.pending.length) return;

      const win = slot.iframe.contentWindow;
      const handler = win && win.ceffy && win.ceffy.onMessageFromUnity;
      if (typeof handler !== "function") {
        waiting = true;
        return;
      }

      const messages = slot.pending.splice(0);
      messages.forEach(function (data) { handler.call(win.ceffy, data); });
    });

    if (!waiting) {
      clearInterval(pendingTimer);
      pendingTimer = 0;
    }
  }

  function deliver(slot, data) {
    const win = slot.iframe.contentWindow;
    const handler = win && win.ceffy && win.ceffy.onMessageFromUnity;
    if (typeof handler !== "function") {
      slot.pending.push(data);
      if (!pendingTimer) pendingTimer = setInterval(flushPending, 50);
      return;
    }
    handler.call(win.ceffy, data);
  }

  function applyLayout(iframe, msg) {
    iframe.style.left = msg.x + "px";
    iframe.style.top = msg.y + "px";
    iframe.style.width = msg.w + "px";
    iframe.style.height = msg.h + "px";
  }

  function addSlot(msg) {
    const id = msg.id;
    removeSlot(id);

    const iframe = document.createElement("iframe");
    iframe.name = "ceffy-slot-" + id;
    iframe.className = "ceffy-slot";
    iframe.setAttribute("allowtransparency", "true");
    iframe.setAttribute("scrolling", "no");
    applyLayout(iframe, msg);

    const slot = { iframe: iframe, pending: [] };
    slots.set(id, slot);

    iframe.addEventListener("load", function () {
      forwardConsole(id, iframe);
      sendToUnity({ type: "loaded", id: id });
    });

    iframe.src = msg.src || "about:blank";
    root.appendChild(iframe);
  }

  function removeSlot(id) {
    const slot = slots.get(id);
    if (!slot) return;
    slot.iframe.remove();
    slots.delete(id);
  }

  function withSlot(id, action) {
    const slot = slots.get(id);
    if (!slot) return;
    try {
      action(slot);
    } catch (err) {
      console.warn("[CeffySharedHost] Slot", id, "failed:", err);
    }
  }

  function blurAll() {
    if (document.activeElement) document.activeElement.blur();
    window.focus();
  }

  function handleUnityMessage(raw) {
    const msg = JSON.parse(raw);
    switch (msg.type) {
      case "add":
        addSlot(msg);
        break;
      case "remove":
        removeSlot(msg.id);
        break;
      case "layout":
        withSlot(msg.id, function (slot) { applyLayout(slot.iframe, msg); });
        break;
      case "navigate":
        withSlot(msg.id, function (slot) {
          slot.pending.length = 0;
          slot.iframe.src = msg.src || "about:blank";
        });
        break;
      case "msg":
        withSlot(msg.id, function (slot) { deliver(slot, msg.data); });
        break;
      case "exec":
        withSlot(msg.id, function (slot) { slot.iframe.contentWindow.eval(msg.data); });
        break;
      case "focus":
        withSlot(msg.id, function (slot) {
          slot.iframe.focus();
          if (slot.iframe.contentWindow) slot.iframe.contentWindow.focus();
        });
        break;
      case "blur":
        blurAll();
        break;
    }
  }

  function wireCeffy() {
    if (!window.ceffy) return false;
    window.ceffy.onMessageFromUnity = handleUnityMessage;
    sendToUnity({ type: "ready" });
    return true;
  }

  if (!wireCeffy()) {
    let tries = 0;
    const timer = setInterval(function () {
      tries++;
      if (wireCeffy() || tries > 100) clearInterval(timer);
    }, 50);
  }
})();
  </script>
</body>
</html>
